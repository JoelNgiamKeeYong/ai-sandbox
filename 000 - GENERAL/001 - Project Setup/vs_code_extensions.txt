VS Code Extensions Set Up

----------------------
✅ Install
----------------------
$ CODE_CMD=$(which code-server) && \
    grep '^- ' vs_code_extensions.txt | sed -n 's/^- *//;s/[[:space:]]*#.*//;s/^[[:space:]]*//;s/[[:space:]]*$//p' | \
    xargs -L 1 "$CODE_CMD" --install-extension

----------------------
❌ Uninstall
----------------------
$ CODE_CMD=$(which code-server) && \
    grep '^- ' vs_code_extensions.txt | sed -n 's/^- *//;s/[[:space:]]*#.*//;s/^[[:space:]]*//;s/[[:space:]]*$//p' | \
    xargs -L 1 "$CODE_CMD" --uninstall-extension
